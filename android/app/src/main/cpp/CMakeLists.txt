cmake_minimum_required(VERSION 3.22.1)
project("almahala_netplay_core")
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_definitions(-D__LIBRETRO__ -DANDROID)

add_library(almahala_netplay_core SHARED almahala_retro_bridge.cpp netplay_sync.cpp combat3_netplay_protocol.cpp)

target_compile_options(almahala_netplay_core PRIVATE -Oz -fvisibility=hidden -fvisibility-inlines-hidden -fdata-sections -ffunction-sections)
set_target_properties(almahala_netplay_core PROPERTIES LINK_FLAGS "-Wl,--gc-sections -Wl,--strip-all")

find_library(log-lib log)
find_library(android-lib android)
find_library(gles3-lib GLESv3)
find_library(egl-lib EGL)
find_library(jnigraphics-lib jnigraphics)
target_link_libraries(almahala_netplay_core ${log-lib} ${android-lib} ${gles3-lib} ${egl-lib} ${jnigraphics-lib})
